\begin{enumerate}[label=\arabic*.]

\item Muestre que la siguiente gramática es $\ll$ pero no $\slr$
  \[
  \begin{aligned}
    \nt{S} &\to \tx{A}\ \tk{a}\ \tx{A}\ \tk{b}\ \mid \tx{B}\ \tk{b}\ \tx{B}\ \tk{a} \\
    \nt{A} &\to \eps \\
    \nt{B} &\to \eps
  \end{aligned}
  \]

  %%%%%%%%%%%%%%%%%%%%%%%%%%
  \bigskip
  %%%%%%%%%%%%%%%%%%%%%%%%%%
  
  \boxed{\textbf{\large P.D.}}\ \textbf{\large La gramática es $\ll$}\\

  \textbf{Paso 1. Anulables}\\
  \nt{A} y \nt{B} son anulables porque tienen la producción $\eps$, pero \nt{S} no es anulable, pues
  cualquiera de sus dos alternativas contiene terminales \tk{a}, \tk{b} que siempre deben generarse.\\

  \textbf{Paso 2. \textsc{First}}\\
  Partimos de lo que no depende de nadie:
  \[
  \First{\tx{A}} = \{\eps\} \qquad \First{\tx{B}} = \{\eps\}
  \]

  Para cada alternativa de \nt{S} recorremos el lado derecho de izquierda a derecha, avanzando mientras el
  símbolo sea anulable:
  \[
  \begin{aligned}
    \First{\tx{A}\,\tk{a}\,\tx{A}\,\tk{b}}
    &= \big(\First{\tx{A}}\setminus\{\eps\}\big) \cup \First{\tk{a}}
    = \varnothing \cup \{\tk{a}\} = \{\tk{a}\} \\
    \First{\tx{B}\,\tk{b}\,\tx{B}\,\tk{a}}
    &= \big(\First{\tx{B}}\setminus\{\eps\}\big) \cup \First{\tk{b}}
    = \varnothing \cup \{\tk{b}\} = \{\tk{b}\}
  \end{aligned}
  \]
  El recorrido se detiene en el terminal porque no es anulable, entonces $\First{\tx{S}}$ es la
  \textbf{unión} de ambas alternativas:
  \[
  \First{\tx{S}} = \{\tk{a}\} \cup \{\tk{b}\} = \{\tk{a},\ \tk{b}\}
  \]

  \textbf{Paso 3. \textsc{Follow}}
  \begin{itemize}
  \item $\Follow{\tx{S}}$: como es el símbolo inicial aporta \tk{\$}, no aparece en ningún lado derecho.
    \[
    \Follow{\tx{S}} = \{\tk{\$}\}
    \]

  \item $\Follow{\tx{A}}$: aparece dos veces en $\nt{S}\to\nt{A}\,\tk{a}\,\nt{A}\,\tk{b}$.\\
    A la primera \nt{A} le sigue $\tk{a}\,\nt{A}\,\tk{b}$, cuyo $\mathrm{FIRST}$ es $\{\tk{a}\}$
    (como ya mostramos).\\
    A la segunda \nt{A} le sigue \tk{b}.\\
    \nt{A} nunca queda al final de una producción, así que no hereda $\Follow{\tx{S}}$.
    \[
    \Follow{\tx{A}} = \{\tk{a},\ \tk{b}\}
    \]

  \item $\Follow{\tx{B}}$: aparece dos veces en $\nt{S}\to\nt{B}\,\tk{b}\,\nt{B}\,\tk{a}$.\\
    A la primera \nt{B} le sigue $\tk{b}\,\nt{B}\,\tk{a}$, que aporta \tk{b}.\\
    A la segunda \nt{B} le sigue \tk{a}.\\
    Tampoco queda al final de ninguna producción.
    \[
    \Follow{\nt{B}} = \{\tk{a},\ \tk{b}\}
    \]
  \end{itemize}

  Al repetir las reglas ningún conjunto vuelve a cambiar, por lo que ya es el punto fijo.\\

  Por lo tanto, los conjuntos \textsc{First} y \textsc{Follow} de la gramática son:
  \[
  \begin{aligned}
    \First{\tx{S}} &= \{\tk{a},\ \tk{b}\} & \Follow{\tx{S}} &= \{\tk{\$}\} \\
    \First{\tx{A}} &= \{\eps\}            & \Follow{\tx{A}} &= \{\tk{a},\ \tk{b}\} \\
    \First{\tx{B}} &= \{\eps\}            & \Follow{\tx{B}} &= \{\tk{a},\ \tk{b}\}
  \end{aligned}
  \]

  \textbf{Paso 4. Producciones enumeradas y su selección}
  \[
  \begin{array}{c l l l}
    \# & \text{Producción} & \text{Colocado en} & \text{Motivo} \\ \hline
    1 & \nt{S}\to\nt{A}\,\tk{a}\,\nt{A}\,\tk{b} & \tk{a} & \First{\tx{A}\,\tk{a}\,\tx{A}\,\tk{b}} \\
    2 & \nt{S}\to\nt{B}\,\tk{b}\,\nt{B}\,\tk{a} & \tk{b} & \First{\tx{B}\,\tk{b}\,\tx{B}\,\tk{a}} \\
    3 & \nt{A}\to\eps & \tk{a},\tk{b} & \Follow{\tx{A}} \\
    4 & \nt{B}\to\eps & \tk{a},\tk{b} & \Follow{\tx{B}}
  \end{array}
  \]

  \textbf{Paso 5. Tabla $\ll$}
  \[
  \begin{array}{c|c|c|c}
    & \tk{a} & \tk{b} & \tk{\$} \\ \hline
    \nt{S} & 1 & 2 &   \\ \hline
    \nt{A} & 3 & 3 &   \\ \hline
    \nt{B} & 4 & 4 &   \\
  \end{array}
  \]

  Ninguna celda tiene más de una producción, por ejemplo para \nt{S}, las dos alternativas tienen
  $\textsc{First}$ disjuntos:
  \[
  \First{\tx{A}\,\tk{a}\,\tx{A}\,\tk{b}} \cap \First{\tx{B}\,\tk{b}\,\tx{B}\,\tk{a}}
  = \{\tk{a}\} \cap \{\tk{b}\} = \varnothing
  \]
  Y \nt{A} y \nt{B} tienen una sola producción cada una, así que no pueden chocar consigo mismas.\\

  Por lo tanto \textbf{La gramática es $\ll$.}

  %%%%%%%%%%%%%%%%%%%%%%%%
  \bigskip
  \bigskip
  %%%%%%%%%%%%%%%%%%%%%%%%
  
  \boxed{\textbf{\large P.D.}}\ \textbf{\large La gramática no es $\slr$}\\

  \textbf{Paso 1. Gramática aumentada}\\
  Agregamos un nuevo símbolo inicial $\nt{S'}$ y se enumeramos las producciones:
  \[
  \begin{aligned}
    (0)\ & \nt{S'} \to \nt{S} \\
    (1)\ & \nt{S}  \to \nt{A}\,\tk{a}\,\nt{A}\,\tk{b} \\
    (2)\ & \nt{S}  \to \nt{B}\,\tk{b}\,\nt{B}\,\tk{a} \\
    (3)\ & \nt{A}  \to \eps \\
    (4)\ & \nt{B}  \to \eps
  \end{aligned}
  \]

  \bigskip
  
  \textbf{Paso 2. Cerrradura y AFD de elementos $\lrc$ (\texttt{goto})}\\
  Una producción $\nt{A}\to\eps$ genera directamente el elemento completo $\nt{A}\to\bullet$, el punto ya está al
  final, por lo que en el estado inicial hay dos reducciones posibles.\\
  \[
  \textbf{closure}\big(\{\nt{S'}\to\bullet\,\nt{S}\}\big)
  \]
  el punto está antes de \nt{S}, se agregan sus dos producciones y en ellas el punto queda antes de \nt{A} y de
  \nt{B}, agregamos $\nt{A}\to\bullet$ y $\nt{B}\to\bullet$.

  \medskip

  \textbf{Estado:}
  \[
  \begin{aligned}
    I_0 = \{\, & \nt{S'} \to \bullet\,\nt{S},\\
               & \nt{S}  \to \bullet\,\nt{A}\,\tk{a}\,\nt{A}\,\tk{b},\\
               & \nt{S}  \to \bullet\,\nt{B}\,\tk{b}\,\nt{B}\,\tk{a},\\
               & \nt{A}  \to \bullet,\\
               & \nt{B}  \to \bullet\,\}
  \end{aligned}
  \]

  \textbf{Transiciones:}
  \[
  \begin{aligned}
    X=\nt{S}:\ \texttt{goto}(I_0,\tx{S}) &= \{\,\nt{S'}\to\nt{S}\,\bullet\,\}
    = \boxed{I_1}\\[3pt]
    X=\nt{A}:\ \texttt{goto}(I_0,\tx{A}) &= \{\,\nt{S}\to\nt{A}\,\bullet\,\tk{a}\,\nt{A}\,\tk{b}\,\}
    = \boxed{I_2}\\[3pt]
    X=\nt{B}:\ \texttt{goto}(I_0,\tx{B}) &= \{\,\nt{S}\to\nt{B}\,\bullet\,\tk{b}\,\nt{B}\,\tk{a}\,\}
    = \boxed{I_3}
  \end{aligned}
  \]

  \medskip

  \textbf{Estado:}\quad $I_1 = \{\,\nt{S'}\to\nt{S}\,\bullet\,\}$\\
  Es estado de aceptación, no tiene transiciones.

  \medskip

  \textbf{Estado:}\quad $I_2 = \{\,\nt{S}\to\nt{A}\,\bullet\,\tk{a}\,\nt{A}\,\tk{b}\,\}$

  \textbf{Transiciones:}
  \[
  \begin{aligned}
    X=\tk{a}:\ \texttt{goto}(I_2,\tk{a}) = \{\, & \nt{S}\to\nt{A}\,\tk{a}\,\bullet\,\nt{A}\,\tk{b},\\
                                                & \nt{A}\to\bullet\,\} = \boxed{I_4}
  \end{aligned}
  \]
  El elemento $\nt{A}\to\bullet$ viene de la cerradura, el punto quedó antes de \nt{A}.

  \medskip

  \textbf{Estado:}\quad $I_3 = \{\,\nt{S}\to\nt{B}\,\bullet\,\tk{b}\,\nt{B}\,\tk{a}\,\}$

  \textbf{Transiciones:}
  \[
  \begin{aligned}
    X=\tk{b}:\ \texttt{goto}(I_3,\tk{b}) = \{\, & \nt{S}\to\nt{B}\,\tk{b}\,\bullet\,\nt{B}\,\tk{a},\\
                                              & \nt{B}\to\bullet\,\} = \boxed{I_7}
  \end{aligned}
  \]

  \medskip

  \textbf{Estado:}
  \[
  \begin{aligned}
    I_4 = \{\, & \nt{S}\to\nt{A}\,\tk{a}\,\bullet\,\nt{A}\,\tk{b},\\
               & \nt{A}\to\bullet\,\}
  \end{aligned}
  \]

  \textbf{Transiciones:}\\
  El elemento $\nt{A}\to\bullet$ ya está completo, no avanza.
  \[
  X=\nt{A}:\ \texttt{goto}(I_4,\tx{A}) = \{\,\nt{S}\to\nt{A}\,\tk{a}\,\nt{A}\,\bullet\,\tk{b}\,\} = \boxed{I_5}
  \]

  \medskip

  \textbf{Estado:}\quad $I_5 = \{\,\nt{S}\to\nt{A}\,\tk{a}\,\nt{A}\,\bullet\,\tk{b}\,\}$

  \textbf{Transiciones:}
  \[
  X=\tk{b}:\ \texttt{goto}(I_5,\tk{b}) = \{\,\nt{S}\to\nt{A}\,\tk{a}\,\nt{A}\,\tk{b}\,\bullet\,\} = \boxed{I_6}
  \]

  \medskip

  \textbf{Estado:}\quad $I_6 = \{\,\nt{S}\to\nt{A}\,\tk{a}\,\nt{A}\,\tk{b}\,\bullet\,\}$\\
  Elemento completo, no tiene transiciones.

  \medskip

  \textbf{Estado:}
  \[
  \begin{aligned}
    I_7 = \{\, & \nt{S}\to\nt{B}\,\tk{b}\,\bullet\,\nt{B}\,\tk{a},\\
               & \nt{B}\to\bullet\,\}
  \end{aligned}
  \]

  \textbf{Transiciones:}
  \[
  X=\nt{B}:\ \texttt{goto}(I_7,\tx{B}) = \{\,\nt{S}\to\nt{B}\,\tk{b}\,\nt{B}\,\bullet\,\tk{a}\,\} = \boxed{I_8}
  \]

  \medskip

  \textbf{Estado:}\quad $I_8 = \{\,\nt{S}\to\nt{B}\,\tk{b}\,\nt{B}\,\bullet\,\tk{a}\,\}$

  \textbf{Transiciones:}
  \[
  X=\tk{a}:\ \texttt{goto}(I_8,\tk{a}) = \{\,\nt{S}\to\nt{B}\,\tk{b}\,\nt{B}\,\tk{a}\,\bullet\,\} = \boxed{I_9}
  \]

  \medskip

  \textbf{Estado:}\quad $I_9 = \{\,\nt{S}\to\nt{B}\,\tk{b}\,\nt{B}\,\tk{a}\,\bullet\,\}$\\
  Elemento completo, no tiene transiciones.\\
  
  \bigskip
  
  \textbf{Paso 3. Tabla $\slr$}\\
  Reglas para llenarla:

  \begin{itemize}
  \item \textbf{desplazar} ($d_N$): solo viene de un \texttt{goto} con terminal.
    Si $\texttt{goto}(I_k,x)=I_N$, entonces $\textbf{accion}\tk{[} k, x \tk{]} = d_N$.
    
  \item \textbf{goto}: si $\texttt{goto}(I_k,\tx{X}) = I_N$ con \nt{X} no terminal,
    entonces $\textbf{goto}\tk{[} k, \tx{X} \tk{]} = N$.
    
  \item \textbf{reducir} ($r_j$): solo para elementos con el punto al final, $\nt{X}\to\alpha\,\bullet$,\
    donde $j$ es el número de la producción. En $\slr$ colocamos únicamente en las columnas de $\Follow{\tx{X}}$.

  \item \textbf{aceptar}: el estado que contiene $\nt{S'}\to\nt{S}\,\bullet$\ lleva \textit{aceptar} en la
    columna \tk{\$}.
  \end{itemize}
  
  Del \textbf{Paso 3} donde mostramos que es $\ll$:\\
  $\Follow{\tx{S}} = \{\tk{\$}\}$ y $\Follow{\tx{A}} = \Follow{\tx{B}} = \{\tk{a}, \tk{b}\}$.
  \[
  \begin{array}{c|c|c|c||c|c|c}
    & \multicolumn{3}{c||}{\textbf{acción}} & \multicolumn{3}{c}{\textbf{goto}} \\ \hline
    \text{Estado} & \tk{a} & \tk{b} & \tk{\$} & \nt{S} & \nt{A} & \nt{B} \\ \hline
    0 & \conf{r3/r4} & \conf{r3/r4} &   & 1 & 2 & 3 \\ \hline
    1 &    &    & \text{aceptar} &   &   &   \\ \hline
    2 & d4 &    &    &   &   &   \\ \hline
    3 &    & d7 &    &   &   &   \\ \hline
    4 & r3 & r3 &    &   & 5 &   \\ \hline
    5 &    & d6 &    &   &   &   \\ \hline
    6 &    &    & r1 &   &   &   \\ \hline
    7 & r4 & r4 &    &   &   & 8 \\ \hline
    8 & d9 &    &    &   &   &   \\ \hline
    9 &    &    & r2 &   &   &   \\
  \end{array}
  \]

  \bigskip
  \bigskip
  
  \textbf{Paso 4. Conflicto}\\
  En $I_0$ hay dos elementos completos $\nt{A}\to\bullet$ y $\nt{B}\to\bullet$.\\
  En $\slr$, cada reducción se coloca en su $\textsc{Follow}$:
  \[
  \Follow{\tx{A}} \cap \Follow{\tx{B}} = \{\tk{a},\tk{b}\} \cap \{\tk{a},\tk{b}\}
  = \{\tk{a},\ \tk{b}\} \neq \varnothing
  \]
  
  Por lo tanto $\textbf{accion}\tk{[} 0, \tk{a} \tk{]} = \{r3,\ r4\}$ y
  $\textbf{accion}\tk{[} 0, \tk{b} \tk{]} = \{r3,\ r4\}$, hay un conflicto en $I_0$ ($r3/r4$), pero los demás estados
  no tienen conflictos.\\

  \textbf{Observación.} $\ll$ sí distingue ambas producciones porque decide con el $\textsc{First}$ de todo el
  lado derecho, y ahí \tk{a} y \tk{b} son distintos. $\slr$, en cambio, debe elegir qué $\eps$ reducir
  usando solo $\textsc{Follow}$, que mezcla todos los contextos donde aparece \nt{A} (o \nt{B}) y los vuelve
  indistinguibles.\\

  De este modo \textbf{La gramática no es $\slr$.}\\

  \text{\large Por lo tanto la gramática es $\ll$ pero no $\slr$.}

\item Muestre que la siguiente gramática es $\slr$ pero no $\ll$
  \[
  \begin{aligned}
    \nt{S} &\to \tx{S}\ \tx{A}\ \mid \tx{A} \\
    \nt{A} &\to \tk{a} 
  \end{aligned}
  \]
 
  %%%%%%%%%%%%%%%%%%%%%%%%%%
  \bigskip
  %%%%%%%%%%%%%%%%%%%%%%%%%%

  \boxed{\textbf{\large P.D.}}\ \textbf{\large La gramática no es $\ll$}\\

  \textbf{Paso 1. Anulables}\\
  Ningún no terminal tiene la producción $\eps$ ni deriva en ella, así que no hay anulables.\\

  \textbf{Paso 2. \textsc{First}}\\
  \[
  \First{\tx{A}} = \{\tk{a}\}
  \]
  Para \nt{S}, la alternativa $\nt{S}\to\nt{S}\,\nt{A}$ empieza con la \nt{S} misma (recursión por la izquierda),
  así que solo aporta $\First{\tx{S}}$, que no agrega nada nuevo.\\
  La alternativa $\nt{S}\to\nt{A}$ aporta $\First{\tx{A}}$:
  \[
  \First{\tx{S}} = \First{\tx{S}} \cup \First{\tx{A}} = \{\tk{a}\}
  \]

  \textbf{Paso 3. \textsc{Follow}}
  \begin{itemize}
  \item $\Follow{\tx{S}}$: es el símbolo inicial, así que aporta \tk{\$}.\\
    En $\nt{S}\to\nt{S}\,\nt{A}$ le sigue \nt{A} que no es anulable, así que se agrega
    $\First{\tx{A}} = \{\tk{a}\}$.
    \[
    \Follow{\tx{S}} = \{\tk{\$},\ \tk{a}\}
    \]

  \item $\Follow{\tx{A}}$: aparece al final de $\nt{S}\to\nt{S}\,\nt{A}$ y de $\nt{S}\to\nt{A}$, así que hereda
    $\Follow{\tx{S}}$ en ambas.
    \[
    \Follow{\tx{A}} = \{\tk{\$},\ \tk{a}\}
    \]
  \end{itemize}

  Por lo tanto, los conjuntos \textsc{First} y \textsc{Follow} de la gramática son:
  \[
  \begin{aligned}
    \First{\tx{S}} &= \{\tk{a}\} & \Follow{\tx{S}} &= \{\tk{\$},\ \tk{a}\} \\
    \First{\tx{A}} &= \{\tk{a}\} & \Follow{\tx{A}} &= \{\tk{\$},\ \tk{a}\}
  \end{aligned}
  \]

  \textbf{Paso 4. Producciones enumeradas y su selección}
  \[
  \begin{array}{c l l l}
    \# & \text{Producción} & \text{Colocado en} & \text{Motivo} \\ \hline
    1 & \nt{S}\to\nt{S}\,\nt{A} & \tk{a} & \First{\tx{S}\,\nt{A}} \\
    2 & \nt{S}\to\nt{A} & \tk{a} & \First{\tx{A}} \\
    3 & \nt{A}\to\tk{a} & \tk{a} & \First{\tk{a}}
  \end{array}
  \]

  \textbf{Paso 5. Tabla $\ll$}
  \[
  \begin{array}{c|c|c}
    & \tk{a} & \tk{\$} \\ \hline
    \nt{S} & \conf{1,\ 2} & \\ \hline
    \nt{A} & 3 & \\
  \end{array}
  \]

  La celda $M\tk{[}\nt{S},\tk{a}\tk{]}$ tiene dos producciones, porque los $\First$ de las dos alternativas de
  \nt{S} no son disjuntos:
  \[
  \First{\tx{S}\,\tx{A}} \cap \First{\tx{A}} = \{\tk{a}\} \cap \{\tk{a}\} = \{\tk{a}\} \neq \varnothing
  \]
  
  La causa es la recursión por la izquierda en \nt{S}, pues ningún analizador descendente con un símbolo de
  anticipación puede decidir cuál alternativa usar viendo solo \tk{a}.\\

  Por lo tanto \textbf{La gramática no es $\ll$.}

  %%%%%%%%%%%%%%%%%%%%%%
  \bigskip
  \bigskip
  %%%%%%%%%%%%%%%%%%%%%%

  \boxed{\textbf{\large P.D.}}\ \textbf{\large La gramática es $\slr$}\\

  \textbf{Paso 1. Gramática aumentada}
  \[
  \begin{aligned}
    (0)\ & \nt{S'} \to \nt{S} \\
    (1)\ & \nt{S} \to \nt{S}\,\nt{A} \\
    (2)\ & \nt{S} \to \nt{A} \\
    (3)\ & \nt{A} \to \tk{a}
  \end{aligned}
  \]
  Además, $\Follow{\tx{S'}} = \{\tk{\$}\}$.\\

  \textbf{Paso 2. Cerradura y AFD de elementos $\lrc$ (\texttt{goto})}\\[4pt]

  \[
  \textbf{closure}\big(\{\nt{S'}\to\bullet\,\nt{S}\}\big)
  \]
  el punto está antes de \nt{S}, se agregan $\nt{S}\to\bullet\,\nt{S}\,\nt{A}$ y $\nt{S}\to\bullet\,\nt{A}$;
  en la segunda el punto queda antes de \nt{A}, se agrega $\nt{A}\to\bullet\,\tk{a}$.

  \medskip

  \textbf{Estado:}
  \[
  \begin{aligned}
    I_0 = \{\, & \nt{S'}\to\bullet\,\nt{S},\\
               & \nt{S}\to\bullet\,\nt{S}\,\nt{A},\\
               & \nt{S}\to\bullet\,\nt{A},\\
               & \nt{A}\to\bullet\,\tk{a}\,\}
  \end{aligned}
  \]

  \textbf{Transiciones:}
  \[
  \begin{aligned}
    X=\nt{S}:\ \texttt{goto}(I_0,\tx{S}) = \{\, & \nt{S'}\to\nt{S}\,\bullet,\\
                                                & \nt{S}\to\nt{S}\,\bullet\,\nt{A},\\
                                                & \nt{A}\to\bullet\,\tk{a}\,\} = \boxed{I_1}
  \end{aligned}
  \]
  
  Al avanzar el punto sobre \nt{S} en $\nt{S}\to\bullet\,\nt{S}\,\nt{A}$ el punto queda antes de \nt{A}, por
  eso la cerradura agrega $\nt{A}\to\bullet\,\tk{a}$.

  \[
  \begin{aligned}
    X=\nt{A}:\ \texttt{goto}(I_0,\tx{A}) &= \{\,\nt{S}\to\nt{A}\,\bullet\,\} = \boxed{I_2}\\[3pt]
    X=\tk{a}:\ \texttt{goto}(I_0,\tk{a}) &= \{\,\nt{A}\to\tk{a}\,\bullet\,\} = \boxed{I_3}
  \end{aligned}
  \]

  \bigskip
  \bigskip

  \textbf{Estado:}
  \[
  \begin{aligned}
    I_1 = \{\, & \nt{S'}\to\nt{S}\,\bullet,\\
               & \nt{S}\to\nt{S}\,\bullet\,\nt{A},\\
               & \nt{A}\to\bullet\,\tk{a}\,\}
  \end{aligned}
  \]
  Contiene $\nt{S'}\to\nt{S}\,\bullet$, así que aquí va \textit{aceptar}.

  \textbf{Transiciones:} $\nt{S'}\to\nt{S}\,\bullet$ ya está completo, no avanza.
  \[
  \begin{aligned}
    X=\nt{A}:\ \texttt{goto}(I_1,\tx{A}) &= \{\,\nt{S}\to\nt{S}\,\nt{A}\,\bullet\,\} = \boxed{I_4}\\[3pt]
    X=\tk{a}:\ \texttt{goto}(I_1,\tk{a}) &= \{\,\nt{A}\to\tk{a}\,\bullet\,\} = I_3
  \end{aligned}
  \]

  \medskip

  \textbf{Estado:}\quad $I_2 = \{\,\nt{S}\to\nt{A}\,\bullet\,\}$\\
  Elemento completo, no tiene transiciones.

  \medskip

  \textbf{Estado:}\quad $I_3 = \{\,\nt{A}\to\tk{a}\,\bullet\,\}$\\
  Elemento completo, no tiene transiciones.

  \medskip

  \textbf{Estado:}\quad $I_4 = \{\,\nt{S}\to\nt{S}\,\nt{A}\,\bullet\,\}$\\
  Elemento completo, no tiene transiciones.\\

  \textbf{Paso 3. Tabla $\slr$}\\
  Reglas para llenarla:
  \begin{itemize}
  \item \textbf{desplazar} ($d_N$): solo viene de un \texttt{goto} con terminal.
    Si $\texttt{goto}(I_k,x) = I_N$, entonces $\textbf{accion}\tk{[} k,x \tk{]} = d_N$.
    
  \item \textbf{goto}: si $\texttt{goto}(I_k, \tx{X}) = I_N$ con \nt{X} no terminal,
    entonces $\textbf{goto}\tk{[} k, \tx{X} \tk{]} = N$.
    
  \item \textbf{reducir} ($r_j$): solo para elementos con el punto al final, $\nt{X}\to\alpha\,\bullet$, donde
    $j$ es el número de la producción. En $\slr$ se coloca únicamente en las columnas de $\Follow{\tx{X}}$.
    
  \item \textbf{aceptar}: el estado que contiene $\nt{S'}\to\nt{S}\,\bullet$ lleva \textit{aceptar} en la
    columna \tk{\$}.
  \end{itemize}

  Del \textbf{Paso 3} de la demostración que no es $\ll$:
  $\Follow{\tx{S}} = \Follow{\tx{A}} = \{\tk{a},\tk{\$}\}$.
  \[
  \begin{array}{c|c|c||c|c}
    & \multicolumn{2}{c||}{\textbf{acción}} & \multicolumn{2}{c}{\textbf{goto}} \\ \hline
    \text{Estado} & \tk{a} & \tk{\$} & \nt{S} & \nt{A} \\ \hline
    0 & d3 &    & 1 & 2 \\ \hline
    1 & d3 & \text{aceptar} &   & 4 \\ \hline
    2 & r2 & r2 &   &   \\ \hline
    3 & r3 & r3 &   &   \\ \hline
    4 & r1 & r1 &   &   \\
  \end{array}
  \]

  \textbf{Paso 4. Verificación de conflictos}\\
  Solo $I_1$ mezcla un elemento completo con un elemento de desplazamiento, $\nt{S'}\to\nt{S}\,\bullet$ se
  acepta en \tk{\$} y $\nt{A}\to\bullet\,\tk{a}$ desplaza con \tk{a}, como son columnas distintas, así que no hay
  conflicto.\\
  $I_2$, $I_3$ e $I_4$ tienen un solo elemento completo y ningún desplazamiento, ninguna celda tiene más de una
  acción.\\

  De este modo \textbf{La gramática es $\slr$}.\\

  \text{\large Por lo tanto es $\slr$ pero no $\ll$.}

  \textbf{\Large Gramática 1}\\

  \textbf{\large a. AFD de elementos $\lrc$}\\

  \textbf{Paso 1. Gramática aumentada y reglas de producción}
  \[
  \begin{aligned}
    (0)\ & \nt{E'} \to \nt{E} \\
    (1)\ & \nt{E} \to \tk{(}\,\nt{L}\,\tk{)} \\
    (2)\ & \nt{E} \to \tk{a} \\
    (3)\ & \nt{L} \to \nt{L}\,\tk{,}\,\nt{E} \\
    (4)\ & \nt{L} \to \nt{E}
  \end{aligned}
  \]

  \textbf{Paso 2. Cerradura y AFD de elementos $\lrc$ (\texttt{goto})}\\
  Cada estado se obtiene con $\texttt{goto}(I_k, \tx{X})$, avanzamos el punto sobre $X$ en los elementos que lo
  tienen justo antes y al resultado le aplicamos la cerradura (si el punto queda antes de un no terminal,
  entonces agregamos sus producciones con el punto al inicio).\\
  \[
  \textbf{closure}\big(\{\nt{E'}\to\bullet\,\nt{E}\}\big)
  \]
  punto está antes de \nt{E}, agregamos $\nt{E}\to\bullet\,\tk{(}\,\nt{L}\,\tk{)}$ y $\nt{E}\to\bullet\,\tk{a}$,
  ambas empiezan con un terminal, así que la cerradura termina.

  \medskip

  \textbf{Estado:}
  \[
  \begin{aligned}
    I_0 = \{\, & \nt{E'}\to\bullet\,\nt{E},\\
               & \nt{E}\to\bullet\,\tk{(}\,\nt{L}\,\tk{)},\\
               & \nt{E}\to\bullet\,\tk{a}\,\}
  \end{aligned}
  \]

  \textbf{Transiciones:}
  \[
  \begin{aligned}
    X=\nt{E}:\ \texttt{goto}(I_0,\tx{E}) = \{\, & \nt{E'}\to\nt{E}\,\bullet\,\} = \boxed{I_1}\\[8pt]
    X=\tk{(}:\ \texttt{goto}(I_0,\tk{(}) = \{\, & \nt{E}\to\tk{(}\,\bullet\,\nt{L}\,\tk{)},\\
                                                & \nt{L}\to\bullet\,\nt{L}\,\tk{,}\,\nt{E},\\
                                                & \nt{L}\to\bullet\,\nt{E},\\
                                                & \nt{E}\to\bullet\,\tk{(}\,\nt{L}\,\tk{)},\\
                                                & \nt{E}\to\bullet\,\tk{a}\,\} = \boxed{I_2}\\[8pt]
    X=\tk{a}:\ \texttt{goto}(I_0,\tk{a}) &= \{\,\nt{E}\to\tk{a}\,\bullet\,\} = \boxed{I_3}
  \end{aligned}
  \]
  
  En $I_2$, tras mover el punto sobre \tk{(} queda antes de \nt{L}, así que agregamos las producciones de \nt{L};
  en $\nt{L}\to\bullet\,\nt{E}$ el punto queda antes de \nt{E}, y agregamos las de \nt{E}.

  \medskip

  \textbf{Estado:}\quad $I_1 = \{\,\nt{E'}\to\nt{E}\,\bullet\,\}$\\
  Es el estado de aceptación, no tiene transiciones.

  \textbf{Estado:}
  \[
  \begin{aligned}
    I_2 = \{\, & \nt{E}\to\tk{(}\,\bullet\,\nt{L}\,\tk{)},\\
               & \nt{L}\to\bullet\,\nt{L}\,\tk{,}\,\nt{E},\\
               & \nt{L}\to\bullet\,\nt{E},\\
               & \nt{E}\to\bullet\,\tk{(}\,\nt{L}\,\tk{)},\\
               & \nt{E}\to\bullet\,\tk{a}\,\}
  \end{aligned}
  \]
  
  \textbf{Transiciones:}
  \[
  \begin{aligned}
    X=\nt{L}:\ \texttt{goto}(I_2,\tx{L}) = \{\, & \nt{E}\to\tk{(}\,\nt{L}\,\bullet\,\tk{)},\\
                                              & \nt{L}\to\nt{L}\,\bullet\,\tk{,}\,\nt{E}\,\} = \boxed{I_4}\\[6pt]
    X=\nt{E}:\ \texttt{goto}(I_2,\tx{E}) &= \{\,\nt{L}\to\nt{E}\,\bullet\,\} = \boxed{I_5}\\[3pt]
    X=\tk{(}:\ \texttt{goto}(I_2,\tk{(}) &= I_2\\[3pt]
    X=\tk{a}:\ \texttt{goto}(I_2,\tk{a}) &= I_3
  \end{aligned}
  \]
  Con \tk{(} se obtiene el mismo conjunto de elementos que $I_2$, por eso regresa a $I_2$. Mientras que con
  \tk{a} se va a $I_3$, pues es la misma siutación $\nt{E}\to\bullet\tk{a}$; a diferencia de con $\nt{E}$ que va
  a $I_5$ que proviene de $\nt{L}\to\bullet\tk{E}$.

  \medskip

  \textbf{Estado:}\quad $I_3 = \{\,\nt{E}\to\tk{a}\,\bullet\,\}$\\
  Elemento completo, no tiene transiciones.

  \medskip

  \textbf{Estado:}
  \[
  \begin{aligned}
    I_4 = \{\, & \nt{E}\to\tk{(}\,\nt{L}\,\bullet\,\tk{)},\\
               & \nt{L}\to\nt{L}\,\bullet\,\tk{,}\,\nt{E}\,\}
  \end{aligned}
  \]

  \textbf{Transiciones:}
  \[
  \begin{aligned}
    X=\tk{)}:\ \texttt{goto}(I_4,\tk{)}) &= \{\,\nt{E}\to\tk{(}\,\nt{L}\,\tk{)}\,\bullet\,\} = \boxed{I_6}\\[6pt]
    X=\tk{,}:\ \texttt{goto}(I_4,\tk{,}) = \{\, & \nt{L}\to\nt{L}\,\tk{,}\,\bullet\,\nt{E},\\
                                              & \nt{E}\to\bullet\,\tk{(}\,\nt{L}\,\tk{)},\\
                                              & \nt{E}\to\bullet\,\tk{a}\,\} = \boxed{I_7}
  \end{aligned}
  \]

  \medskip

  \textbf{Estado:}\quad $I_5 = \{\,\nt{L}\to\nt{E}\,\bullet\,\}$\\
  Elemento completo, no tiene transiciones.

  \medskip

  \textbf{Estado:}\quad $I_6 = \{\,\nt{E}\to\tk{(}\,\nt{L}\,\tk{)}\,\bullet\,\}$\\
  Elemento completo, no tiene transiciones.

  \medskip

  \textbf{Estado:}
  \[
  \begin{aligned}
    I_7 = \{\, & \nt{L}\to\nt{L}\,\tk{,}\,\bullet\,\nt{E},\\

  \textbf{\large b. Tabla de análisis sintáctico $\slr$}\\

  \textbf{Paso 1. Anulables}\\
  No hay producciones $\eps$, así que ningún no terminal es anulable.\\

  \textbf{Paso 2. \textsc{First}}\\
  $\nt{E}\to\tk{(}\,\nt{L}\,\tk{)}$ aporta \tk{(} y $\nt{E}\to\tk{a}$ aporta \tk{a}:
  \[
  \First{\tx{E}} = \{\tk{(},\ \tk{a}\}
  \]
  Para \nt{L}, la alternativa $\nt{L}\to\nt{L}\,\tk{,}\,\nt{E}$ empieza con \nt{L} misma (recursión por la
  izquierda) y no agrega nada nuevo, la alternativa $\nt{L}\to\nt{E}$ aporta $\First{\tx{E}}$:
  \[
  \First{\tx{L}} = \First{\tx{E}} = \{\tk{(},\ \tk{a}\}
  \]

  \textbf{Paso 3. \textsc{Follow}}
  \begin{itemize}
  \item $\Follow{\tx{L}}$: en $\nt{E}\to\tk{(}\,\nt{L}\,\tk{)}$ le sigue \tk{)} y en

    \[
    \Follow{\tx{L}} = \{\tk{)},\ \tk{,}\}
    \]

  \item $\Follow{\tx{E}}$: es el símbolo inicial, así que aporta \tk{\$}. En $\nt{L}\to\nt{L}\,\tk{,}\,\nt{E}$

    \[
    \Follow{\tx{E}} = \{\tk{\$}\} \cup \Follow{\tx{L}} = \{\tk{\$},\ \tk{)},\ \tk{,}\}

    $\Follow{\tx{X}}$.

  \textbf{\large d. ¿La gramática 1 es $\lrc$?}\\

  Una gramática es $\lrc$ si ningún estado del AFD tiene un conflicto $d_N/r_J$ (un elemento completo junto con
  un elemento con un terminal después del punto) o $r_J/r_{J'}$ (dos elementos completos).\\
  Revisando cada estado:
  \begin{itemize}
  \item $I_0$, $I_2$, $I_7$: solo elementos con el punto antes de un símbolo, sin elementos completos.
  \item $I_1$: solo $\nt{E'}\to\nt{E}\,\bullet$, que es \textit{aceptar} en la columna \tk{\$}.
  \item $I_4$: sus dos elementos tienen el punto antes de \tk{)} y antes de \tk{,}. Solo hay desplazamientos.
  \item $I_3$, $I_5$, $I_6$, $I_8$: un único elemento completo cada uno.
  \end{itemize}
  
  No hay ningún conflicto, por lo tanto \textbf{la gramática 1 es $\lrc$}. En $\lrc$ una reducción se coloca en
  \textbf{todas} las columnas de terminales (sin usar $\mathrm{FOLLOW}$), así:
  \[
  \begin{array}{c|c|c|c|c|c||c|c}
    & \multicolumn{5}{c||}{\textbf{acción}} & \multicolumn{2}{c}{\textbf{goto}} \\ \hline

    2 & d2 &    & d3 &    &    & 5 & 4 \\ \hline

    4 &    & d6 &    & d7 &    &   &   \\ \hline

    6 & r1 & r1 & r1 & r1 & r1 &   &   \\ \hline
    7 & d2 &    & d3 &    &    & 8 &   \\ \hline
    8 & r3 & r3 & r3 & r3 & r3 &   &   \\
  \end{array}
  \]

  \textbf{\Large Gramática 2}\\

  \textbf{\large a. AFD de elementos $\lrc$}\\

  \textbf{Paso 1. Gramática aumentada y reglas de producción}
  \[
  \begin{aligned}
    (0)\ & \nt{A'} \to \nt{A} \\
    (1)\ & \nt{A} \to \nt{A}\,\tk{(}\,\nt{A}\,\tk{)} \\
    (2)\ & \nt{A} \to \eps
  \end{aligned}
  \]

  \textbf{Paso 2. Cerradura y AFD de elementos $\lrc$ (\texttt{goto})}\\
  La producción $\nt{A}\to\eps$ genera directamente el elemento completo $\nt{A}\to\bullet$.\\

  \[
  \textbf{closure}\big(\{\nt{A'}\to\bullet\,\nt{A}\}\big)
  \]
  punto está antes de \nt{A}, agregamos $\nt{A}\to\bullet\,\nt{A}\,\tk{(}\,\nt{A}\,\tk{)}$ y $\nt{A}\to\bullet$.

  \medskip

  \textbf{Estado:}
  \[
  \begin{aligned}
    I_0 = \{\, & \nt{A'}\to\bullet\,\nt{A},\\
               & \nt{A}\to\bullet\,\nt{A}\,\tk{(}\,\nt{A}\,\tk{)},\\
               & \nt{A}\to\bullet\,\}
  \end{aligned}
  \]

  \textbf{Transiciones:} el elemento $\nt{A}\to\bullet$ ya está completo, no avanza.
  \[
  \begin{aligned}
    X=\nt{A}:\ \texttt{goto}(I_0,\tx{A}) = \{\, & \nt{A'}\to\nt{A}\,\bullet,\\
                                            & \nt{A}\to\nt{A}\,\bullet\,\tk{(}\,\nt{A}\,\tk{)}\,\} = \boxed{I_1}
  \end{aligned}
  \]

  \medskip

  \textbf{Estado:}
  \[

  Para este ejercicio, debemos definir lo que es el \textbf{lookahead} (símbolo de anticipación, concepto que
  usaremos tanto para $\lr$ y $\lalr$), este consiste en el siguiente terminal de la cadena de entrada, que el
  analizador mira sin consumirlo para decidir qué acción realizar. El 1 de $\lr$ y $\lalr$ indica que
  se usa un símbolo de anticipación, en $\lr$, cada elemento lleva su propio lookahead:
  \[
  [\nt{X}\to\alpha\,\bullet\,\beta,\ x]
  \]
  donde $x$ es un terminal (o \tk{\$}, el fin de la cadena) que debe aparecer inmediatamente después de
  $\nt{X}\to\alpha\beta$ cuando esta producción se reduce. Por eso, un elemento con el punto al final,
  $[\nt{X}\to\alpha\,\bullet,\ x]$, solo nos permite reducir cuando el siguiente símbolo de la entrada es $x$.\\

  Los lookaheads se calculan así:
  \begin{itemize}
  \item El elemento inicial es $[\nt{S'}\to\bullet\,\nt{S},\ \tk{\$}]$, pues después de la cadena completa solo
    puede venir el fin de la entrada.
    
  \item En la cerradura, el lookahead de lo que se agrega es $\First{\beta\,x}$, es decir, lo que puede seguir
    a \nt{Y} en ese contexto. Por ejemplo, en la gramática 1, de $[\nt{E'}\to\bullet\,\nt{E},\ \tk{\$}]$ no hay
    nada después de \nt{E} ($\beta=\eps$), así que $[\nt{E}\to\bullet\,\tk{a},\ y]$ con
    $y\in\First{\tk{\$}}=\{\tk{\$}\}$. En cambio, de $[\nt{E}\to\tk{(}\,\bullet\,\nt{L}\,\tk{)},\ \tk{\$}]$
    después de \nt{L} viene \tk{)}, así que
    $[\nt{L}\to\bullet\,\nt{E},\ y]$ con $y\in\First{\tk{)}\,\tk{\$}}=\{\tk{)}\}$.
    
  \item En \textbf{goto}, el elemento conserva su lookahead al avanzar el punto.
  \end{itemize}

  \bigskip

  \textbf{Reglas $\lr$ que se usan.}\\
  Un elemento es $[\nt{X}\to\alpha\,\bullet\,\beta,\ x]$, con $x$ el lookahead.
  \begin{itemize}
  \item \textbf{Cerradura}: de $[\nt{X}\to\alpha\,\bullet\,\nt{Y}\,\beta,\ x]$ agregamos
    $[\nt{Y}\to\bullet\,\gamma,\ y]$ para cada producción de \nt{Y} y cada $y\in\First{\beta\,x}$.

  \item \textbf{\texttt{goto}($I_k$, $X$)}: avanzamos el punto sobre $X$ conservando el \textbf{lookahead} de
    cada elemento y aplicamos la cerradura.
    
  \item \textbf{Tabla}: \textbf{desplazar} y \textbf{goto} igual que en $\slr$; una reducción
    $[\nt{X}\to\alpha\,\bullet,\ y]$ va solo en la columna $y$; \textbf{aceptar} en la columna \tk{\$} del
    estado con $[\nt{S'}\to\nt{S}\,\bullet,\ \tk{\$}]$.
    
  \item $\lalr$: fusionamos los estados del AFD $\lr$ con el mismo núcleo (mismos elementos sin
    \textbf{lookaheads}), uniendo los \textbf{lookaheads} de cada elemento.
  \end{itemize}

  Un lookahead escrito $\tk{)}/\tk{,}$ significa que se tiene un elemento por cada terminal. Es decir,
  $[\nt{X}\to\alpha\,\bullet\,\beta,\ \tk{)}/\tk{,}]$ abrevia $[\nt{X}\to\alpha\,\bullet\,\beta,\ \tk{)}]$ y
  $[\nt{X}\to\alpha\,\bullet\,\beta,\ \tk{,}]$.\\

  \[
  \begin{aligned}
    X=\nt{A}:\ \texttt{goto}(I_2,\tx{A}) &= \{\,[\nt{S}\to\tk{a}\,\nt{A}\,\bullet\,\tk{d},\ \tk{\$}]\,\} = \boxed{I_4}\\[3pt]
    X=\nt{B}:\ \texttt{goto}(I_2,\tx{B}) &= \{\,[\nt{S}\to\tk{a}\,\nt{B}\,\bullet\,\tk{e},\ \tk{\$}]\,\} = \boxed{I_5}\\[3pt]
    X=\tk{c}:\ \texttt{goto}(I_2,\tk{c}) &= \{\,[\nt{A}\to\tk{c}\,\bullet,\ \tk{d}],\ [\nt{B}\to\tk{c}\,\bullet,\ \tk{e}]\,\} = \boxed{I_6}
  \end{aligned}
  \]

  \medskip

  \textbf{Estado:}
  \[
  \begin{aligned}
  I_3 = \{\, & [\nt{S}\to\tk{b}\,\bullet\,\nt{B}\,\tk{d},\ \tk{\$}],\\
             & [\nt{S}\to\tk{b}\,\bullet\,\nt{A}\,\tk{e},\ \tk{\$}],\\
             & [\nt{B}\to\bullet\,\tk{c},\ \tk{d}],\\
             & [\nt{A}\to\bullet\,\tk{c},\ \tk{e}]\,\}
  \end{aligned}
  \]
  
  \textbf{Transiciones:}
  \[
  \begin{aligned}
    X=\nt{B}:\ \texttt{goto}(I_3,\nt{B}) &= \{\,[\nt{S}\to\tk{b}\,\nt{B}\,\bullet\,\tk{d},\ \tk{\$}]\,\} = \boxed{I_7}\\[3pt]
    X=\nt{A}:\ \texttt{goto}(I_3,\nt{A}) &= \{\,[\nt{S}\to\tk{b}\,\nt{A}\,\bullet\,\tk{e},\ \tk{\$}]\,\} = \boxed{I_8}\\[3pt]
    X=\tk{c}:\ \texttt{goto}(I_3,\tk{c}) &= \{\,[\nt{B}\to\tk{c}\,\bullet,\ \tk{d}],\ [\nt{A}\to\tk{c}\,\bullet,\ \tk{e}]\,\} = \boxed{I_9}
  \end{aligned}
  \]

  \medskip

  \textbf{Estados:}
  \[
  \begin{aligned}
    I_4 &= \{\,[\nt{S}\to\tk{a}\,\nt{A}\,\bullet\,\tk{d},\ \tk{\$}]\,\} &
      \texttt{goto}(I_4,\tk{d}) &= \{\,[\nt{S}\to\tk{a}\,\nt{A}\,\tk{d}\,\bullet,\ \tk{\$}]\,\} = \boxed{I_{10}}\\[3pt]
    I_5 &= \{\,[\nt{S}\to\tk{a}\,\nt{B}\,\bullet\,\tk{e},\ \tk{\$}]\,\} &
      \texttt{goto}(I_5,\tk{e}) &= \{\,[\nt{S}\to\tk{a}\,\nt{B}\,\tk{e}\,\bullet,\ \tk{\$}]\,\} = \boxed{I_{11}}\\[3pt]
    I_7 &= \{\,[\nt{S}\to\tk{b}\,\nt{B}\,\bullet\,\tk{d},\ \tk{\$}]\,\} &
      \texttt{goto}(I_7,\tk{d}) &= \{\,[\nt{S}\to\tk{b}\,\nt{B}\,\tk{d}\,\bullet,\ \tk{\$}]\,\} = \boxed{I_{12}}\\[3pt]
    I_8 &= \{\,[\nt{S}\to\tk{b}\,\nt{A}\,\bullet\,\tk{e},\ \tk{\$}]\,\} &
      \texttt{goto}(I_8,\tk{e}) &= \{\,[\nt{S}\to\tk{b}\,\nt{A}\,\tk{e}\,\bullet,\ \tk{\$}]\,\} = \boxed{I_{13}}
  \end{aligned}
  \]

  \medskip

  \textbf{Estados con elementos completos}, es decir, no tienen transiciones:
  \[
  \begin{aligned}
    I_6 &= \{\,[\nt{A}\to\tk{c}\,\bullet,\ \tk{d}],\ [\nt{B}\to\tk{c}\,\bullet,\ \tk{e}]\,\} &
    I_9 &= \{\,[\nt{B}\to\tk{c}\,\bullet,\ \tk{d}],\ [\nt{A}\to\tk{c}\,\bullet,\ \tk{e}]\,\}\\[3pt]
    I_{10} &= \{\,[\nt{S}\to\tk{a}\,\nt{A}\,\tk{d}\,\bullet,\ \tk{\$}]\,\} &
    I_{11} &= \{\,[\nt{S}\to\tk{a}\,\nt{B}\,\tk{e}\,\bullet,\ \tk{\$}]\,\}\\[3pt]
    I_{12} &= \{\,[\nt{S}\to\tk{b}\,\nt{B}\,\tk{d}\,\bullet,\ \tk{\$}]\,\} &
    I_{13} &= \{\,[\nt{S}\to\tk{b}\,\nt{A}\,\tk{e}\,\bullet,\ \tk{\$}]\,\}
  \end{aligned}
  \]

  \textbf{Paso 3. Tabla $\lr$}\\
  Reglas para llenarla:
  \begin{itemize}
  \item \textbf{desplazar} ($d_N$): solo viene de un \texttt{goto} con terminal. Si
    $\texttt{goto}(I_k,x) = I_N$, entonces $\textbf{accion}\tk{[} k, x \tk{}] = d_N$.
    
  \item \textbf{goto}: si $\texttt{goto}(I_k, \tx{X}) = I_N$ con \nt{X} no terminal, entonces
    $\textbf{goto} \tk{[} k, \tx{X} \tk{]} = N$.
    
  \item \textbf{reducir} ($r_j$): para cada elemento $[\nt{X}\to\alpha\,\bullet,\ y]$, donde $j$ es el
    número de la producción. En $\lr$ se coloca \textbf{únicamente en la columna $y$}, el lookahead del elemento.
    
  \item \textbf{aceptar}: el estado que contiene $[\nt{S'}\to\nt{S}\,\bullet,\ \tk{\$}]$ lleva \textit{aceptar}
    en la columna \tk{\$}.
  \end{itemize}

  \[
  \begin{array}{c|c|c|c|c|c|c||c|c|c}
    & \multicolumn{6}{c||}{\textbf{acción}} & \multicolumn{3}{c}{\textbf{goto}} \\ \hline
    \text{Estado} & \tk{a} & \tk{b} & \tk{c} & \tk{d} & \tk{e} & \tk{\$} & \nt{S} & \nt{A} & \nt{B} \\ \hline
    0  & d2 & d3 &     &     &     &     & 1 &   &   \\ \hline
    1  &    &    &     &     &     & \text{aceptar} &   &   &   \\ \hline
    2  &    &    & d6  &     &     &     &   & 4 & 5 \\ \hline
    3  &    &    & d9  &     &     &     &   & 8 & 7 \\ \hline
    4  &    &    &     & d10 &     &     &   &   &   \\ \hline
    5  &    &    &     &     & d11 &     &   &   &   \\ \hline
    6  &    &    &     & r5  & r6  &     &   &   &   \\ \hline
    7  &    &    &     & d12 &     &     &   &   &   \\ \hline
    8  &    &    &     &     & d13 &     &   &   &   \\ \hline
    9  &    &    &     & r6  & r5  &     &   &   &   \\ \hline
    10 &    &    &     &     &     & r1  &   &   &   \\ \hline
    11 &    &    &     &     &     & r3  &   &   &   \\ \hline
    12 &    &    &     &     &     & r2  &   &   &   \\ \hline
    13 &    &    &     &     &     & r4  &   &   &   \\
  \end{array}
  \]

  \textbf{Paso 4. Verificación de conflictos}\\
  Los únicos estados donde podría haber conflicto son $I_6$ e $I_9$, que tienen dos elementos completos:
  \begin{itemize}
  \item $I_6$: $\nt{A}\to\tk{c}\,\bullet$ con lookahead $\{\tk{d}\}$ y $\nt{B}\to\tk{c}\,\bullet$ con lookahead
    $\{\tk{e}\}$, son disjuntos.
    
  \item $I_9$: $\nt{B}\to\tk{c}\,\bullet$ con lookahead $\{\tk{d}\}$ y $\nt{A}\to\tk{c}\,\bullet$ con lookahead
    $\{\tk{e}\}$, también son disjuntos.
  \end{itemize}
  El resto de los estados tiene un solo elemento completo o solo elementos de desplazamiento.
  Ninguna celda tiene más de una acción.\\

  \textbf{La gramática es $\lr$.}

  \bigskip
  %%%%%%%%%%%%%%%%%%%%%%%%%%%%

  \boxed{\textbf{\large P.D.}}\ \textbf{\large La gramática no es $\lalr$}\\

  \textbf{Paso 1. Estados con el mismo núcleo}\\
  El núcleo de un estado son sus elementos sin lookaheads. Comparando $I_6$ e $I_9$:
  \[
  \begin{aligned}
    I_6 &= \{[\nt{A}\to\tk{c}\,\bullet,\ \tk{d}],\ [\nt{B}\to\tk{c}\,\bullet,\ \tk{e}]\} \\
    I_9 &= \{[\nt{B}\to\tk{c}\,\bullet,\ \tk{d}],\ [\nt{A}\to\tk{c}\,\bullet,\ \tk{e}]\}
  \end{aligned}
  \]
  Ambos tienen el mismo núcleo $\{\nt{A}\to\tk{c}\,\bullet,\ \nt{B}\to\tk{c}\,\bullet\}$. $I_2$ e $I_3$ no se
  fusionan porque sus núcleos son distintos (en $I_2$ aparecen $\nt{S}\to\tk{a}\,\bullet\ldots$ y en $I_3$
  $\nt{S}\to\tk{b}\,\bullet\ldots$). Ningún otro par de estados comparte núcleo.\\

  \textbf{Paso 2. Fusión y AFD $\lalr$}\\
  Se fusionan $I_6$ e $I_9$ en un solo estado, uniendo los lookaheads de cada elemento:
  \[
  I_{69} = I_6 \cup I_9 = \{\,[\nt{A}\to\tk{c}\,\bullet,\ \tk{d}/\tk{e}],\ [\nt{B}\to\tk{c}\,\bullet,\ \tk{d}/\tk{e}]\,\}
  \]
  Los demás estados quedan igual, lo único que cambia son las transiciones con \tk{c}, que ahora llegan al
  estado fusionado:
  \[
  \begin{aligned}
    X=\tk{c}:\ \texttt{goto}(I_2,\tk{c}) &= \boxed{I_{69}}\\[3pt]
    X=\tk{c}:\ \texttt{goto}(I_3,\tk{c}) &= \boxed{I_{69}}
  \end{aligned}
  \]
  El AFD $\lalr$ tiene 13 estados: $I_0,\ldots,I_5,\ I_{69},\ I_7,\ I_8,\ I_{10},\ldots,I_{13}$.\\

  \textbf{Paso 3. Tabla $\lalr$}
  \[
  \begin{array}{c|c|c|c|c|c|c||c|c|c}
    & \multicolumn{6}{c||}{\textbf{acción}} & \multicolumn{3}{c}{\textbf{goto}} \\ \hline
    \text{Estado} & \tk{a} & \tk{b} & \tk{c} & \tk{d} & \tk{e} & \tk{\$} & \nt{S} & \nt{A} & \nt{B} \\ \hline
    0  & d2 & d3 &      &     &     &     & 1 &   &   \\ \hline
    1  &    &    &      &     &     & \text{aceptar} &   &   &   \\ \hline
    2  &    &    & d69  &     &     &     &   & 4 & 5 \\ \hline
    3  &    &    & d69  &     &     &     &   & 8 & 7 \\ \hline
    4  &    &    &      & d10 &     &     &   &   &   \\ \hline
    5  &    &    &      &     & d11 &     &   &   &   \\ \hline
    69 &    &    &      & \conf{r5/r6} & \conf{r5/r6} &     &   &   &   \\ \hline
    7  &    &    &      & d12 &     &     &   &   &   \\ \hline
    8  &    &    &      &     & d13 &     &   &   &   \\ \hline
    10 &    &    &      &     &     & r1  &   &   &   \\ \hline
    11 &    &    &      &     &     & r3  &   &   &   \\ \hline
    12 &    &    &      &     &     & r2  &   &   &   \\ \hline
    13 &    &    &      &     &     & r4  &   &   &   \\
  \end{array}
  \]

  \textbf{Paso 4. Conflicto}\\
  En $I_{69}$ ambos elementos completos tienen ahora los mismos lookaheads $\{\tk{d},\tk{e}\}$, así que
  $\textbf{accion}\tk{[} 69,\ tk{d} \tk{]} = \{r5,\ r6\}$ y
  $\textbf{accion} \tk{[} 69, \tk{e} \tk{]} = \{r5,\ r6\}$, hay un conflicto $r5/r6$.\\
  Al tener \tk{c} en el tope y \tk{d} (o \tk{e}) como siguiente símbolo, el analizador $\lalr$ ya no recuerda
  si venía de \tk{a} o de \tk{b}, que es lo que decidía entre \nt{A} y \nt{B}.\\

  \textbf{Observación.} La gramática tampoco es $\slr$, pues
  $\Follow{\tx{A}} = \Follow{\tx{B}} = \{\tk{d},\tk{e}\}$ produce el mismo conflicto. Fusionar estados no
  puede crear conflictos $d_N/r_j$ (el desplazamiento depende solo del núcleo), únicamente
  $r_j/r_{j'}$, y eso es lo que ocurre en este caso.\\

  Así \textbf{La gramática no es $\lalr$}.\\

  \text{\large Por lo tanto es $\lr$ pero no $\lalr$.}
  
\end{enumerate}
